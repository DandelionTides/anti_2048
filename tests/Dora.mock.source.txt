export const json = {
  encode(value: object): [string | undefined, string | undefined] {
    try { return [JSON.stringify(value), undefined]; }
    catch (error) { return [undefined, String(error)]; }
  },
  decode(value: string): [unknown, string | undefined] {
    try { return [JSON.parse(value), undefined]; }
    catch (error) { return [undefined, String(error)]; }
  },
};
